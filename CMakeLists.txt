cmake_minimum_required(VERSION 3.20)
project(streamlink VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
if(MSVC)
  add_compile_options(/utf-8 /W4)
endif()
add_executable(streamlink legacy/src/main.cpp)
target_include_directories(streamlink PRIVATE shared/include)
if(WIN32)
  target_link_libraries(streamlink PRIVATE ws2_32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(streamlink PRIVATE Threads::Threads)
endif()
enable_testing()
add_test(NAME protocol_and_phy COMMAND streamlink selftest)
option(STREAMLINK_UHD "Build experimental X310 radio bridge (requires UHD SDK)" OFF)
add_library(video_radio STATIC shared/radio_device.cpp)
target_include_directories(video_radio PUBLIC shared/include)
target_link_libraries(video_radio PUBLIC Threads::Threads)
add_executable(video_transmitter transmitter/main.cpp transmitter/transmitter_app.cpp)
add_executable(video_receiver receiver/main.cpp receiver/receiver_app.cpp)
target_link_libraries(video_transmitter PRIVATE video_radio)
target_link_libraries(video_receiver PRIVATE video_radio)
if(WIN32)
  target_link_libraries(video_transmitter PRIVATE ws2_32 shell32)
  target_link_libraries(video_receiver PRIVATE ws2_32 shell32)
endif()
if(STREAMLINK_UHD)
  find_package(UHD REQUIRED)
  target_compile_definitions(video_radio PRIVATE STREAMLINK_HAS_UHD=1)
  target_include_directories(video_radio PRIVATE ${UHD_INCLUDE_DIRS})
  target_link_libraries(video_radio PUBLIC ${UHD_LIBRARIES})
  add_executable(streamlink_usrp legacy/src/usrp.cpp)
  target_include_directories(streamlink_usrp PRIVATE shared/include ${UHD_INCLUDE_DIRS})
  target_link_libraries(streamlink_usrp PRIVATE ${UHD_LIBRARIES})
  target_link_libraries(streamlink_usrp PRIVATE Threads::Threads)
  if(WIN32)
    target_link_libraries(streamlink_usrp PRIVATE ws2_32)
  endif()
endif()
